COX_landreform_startup = {
    
    COX = {
        #country flags
        set_country_flag = COX_landreform_ongoing

        #full scope variables
        set_variable = { COX_landreform_progress = 0 }

        set_variable = { COX_landreform_militia = 0 }

        set_variable = { COX_landreform_militia_change = 0 }
    
        set_variable = { COX_landreform_selected_region = 617 }

        set_variable = { COX_landreform_current_action_id = 0 }

        set_variable = { COX_landreform_view_selected = 0 }

        #arrays
        clear_array = COX_landreform_regions_array #土改地区数组
        add_to_array = { COX_landreform_target_array = 617 } #中央
        add_to_array = { COX_landreform_target_array = 957 } #东疆
        add_to_array = { COX_landreform_target_array = 618 } #北疆
        add_to_array = { COX_landreform_target_array = 619 } #南疆


        clear_array = COX_landreform_dec_all_array #土改决议后端数组，用来存储所有潜在决议
        #101-117 中央决议
        for_loop_effect = {
            start = 101
            end = 118
            value = i

            add_to_array = { 
                  array = COX_landreform_dec_all_array
                  value= i 
            }
        }

        #201-227 东疆决议
        for_loop_effect = {
            start = 201
            end = 228
            value = i

            add_to_array = { 
                  array = COX_landreform_dec_all_array
                  value= i 
            }
        }

        #301-321 北疆决议
        for_loop_effect = {
            start = 301
            end = 322
            value = i

            add_to_array = { 
                  array = COX_landreform_dec_all_array
                  value= i 
            }
        }

        #401-426 南疆决议
        for_loop_effect = {
            start = 401
            end = 427
            value = i

            add_to_array = { 
                  array = COX_landreform_dec_all_array
                  value= i 
            }
        }

        clear_array = COX_landreform_dec_done_array #土改决议后端数组，用来存储所有已执行决议
        
    }
    #地区变量初始化
    617 = {#中央
        set_variable = { COX_landreform_local_progress = 0 }
        set_variable = { COX_landreform_local_infrastructure = 40 }
        set_variable = { COX_landreform_local_infrastructure_expectation = 75 }
        set_variable = { COX_landreform_local_unrest = 15 }
        set_variable = { COX_landreform_local_unrest_change = 0 }
        set_variable = { COX_landreform_local_support = 20 }
        set_variable = { COX_landreform_local_support_increase_factor = 0.1 }
        set_variable = { COX_landreform_local_support_decrease_factor = 0.2 }
        set_variable = { COX_landreform_local_militia_demand = 6000 }
    }

    957 = {#东疆
        set_variable = { COX_landreform_local_progress = 0 }
        set_variable = { COX_landreform_local_infrastructure = 20 }
        set_variable = { COX_landreform_local_infrastructure_expectation = 40 }
        set_variable = { COX_landreform_local_unrest = 25 }
        set_variable = { COX_landreform_local_unrest_change = 0 }
        set_variable = { COX_landreform_local_support = 10 }
        set_variable = { COX_landreform_local_support_increase_factor = 0.05 }
        set_variable = { COX_landreform_local_support_decrease_factor = 0.1 }
        set_variable = { COX_landreform_local_militia_demand = 9000 }
    }

    618 = {#北疆
        set_variable = { COX_landreform_local_progress = 0 }
        set_variable = { COX_landreform_local_infrastructure = 15 }
        set_variable = { COX_landreform_local_infrastructure_expectation = 45 }
        set_variable = { COX_landreform_local_unrest = 30 }
        set_variable = { COX_landreform_local_unrest_change = 5 }
        set_variable = { COX_landreform_local_support = 30 }
        set_variable = { COX_landreform_local_support_increase_factor = 0.25 }
        set_variable = { COX_landreform_local_support_decrease_factor = 0.5 }
        set_variable = { COX_landreform_local_militia_demand = 6000 }
    }

    619 = {#南疆
        set_variable = { COX_landreform_local_progress = 0 }
        set_variable = { COX_landreform_local_infrastructure = 10 }
        set_variable = { COX_landreform_local_infrastructure_expectation = 40 }
        set_variable = { COX_landreform_local_unrest = 25 }

#笔记：
#土改决议定义由三部分构成：
#COX_landreform_dec_[决议id]_effect，为执行效果
#位于下方
#COX_landreform_dec_[决议id]_avaliable_trigger，为可用条件
#COX_landreform_dec_[决议id]_visible_trigger，为可见条件
#此二者位于common/scripted_triggers/LFB_landreform_triggers文件中，分别有示例
#决议id为三位数字，其中第一位取[1,2,3,4]，代表地区
#101-117为中央决议，201-227为东疆决议，301-321为北疆决议，401-426为南疆决议

COX_landreform_update = {
    for_each_scope_loop = {
        array = COX_landreform_regions_array
        if = {
            limit = { has_state_flag = COX_landreform_update_avaliable }
            COX_landreform_update_calc = yes
        }
    }
    COX_landreform_refresh = yes
    
}

COX_landreform_update_calc = {
    add_to_variable = {
        COX_landreform_local_unrest = COX_landreform_local_unrest_change
    }
    if = {
        limit = {
            check_variable = { COX_landreform_local_infrastructure > COX_landreform_local_infrastructure_expectation }
        }
        set_temp_variable = {
            temp_COX_landreform_local_support_change = COX_landreform_local_infrastructure_expectation
        }
        subtract_from_temp_variable= {
            temp_COX_landreform_local_support_change = COX_landreform_local_infrastructure
        }
        multiply_temp_variable = {
            temp_COX_landreform_local_support_change = COX_landreform_local_support_increase_factor
        }
        add_to_variable = {
            COX_landreform_local_support = temp_COX_landreform_local_support_change
        }
    }
    else_if = {
        limit = {
            check_variable = { COX_landreform_local_infrastructure < COX_landreform_local_infrastructure_expectation }
        }
        set_temp_variable = {
            temp_COX_landreform_local_support_change = COX_landreform_local_infrastructure
        }
        subtract_from_temp_variable= {
            temp_COX_landreform_local_support_change = COX_landreform_local_infrastructure_expectation
        }
        multiply_temp_variable = {
            temp_COX_landreform_local_support_change = COX_landreform_local_support_decrease_factor
        }
        add_to_variable = {
            COX_landreform_local_support = temp_COX_landreform_local_support_change
        }
    }
}

        }

            }
