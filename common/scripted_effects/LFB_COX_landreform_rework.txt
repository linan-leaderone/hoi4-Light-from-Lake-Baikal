COX_landreform_update = {
    for_each_scope_loop = {
        array = COX_landreform_regions_array
        if = {
            limit = { has_state_flag = COX_landreform_update_avaliable }
            COX_landreform_update_calc = yes
        }
    }
    COX_landreform_refresh = yes
    
}

COX_landreform_update_calc = {
    add_to_variable = {
        COX_landreform_local_unrest = COX_landreform_local_unrest_change
    }
    if = {
        limit = {
            check_variable = { COX_landreform_local_infrastructure > COX_landreform_local_infrastructure_expectation }
        }
        set_temp_variable = {
            temp_COX_landreform_local_support_change = COX_landreform_local_infrastructure_expectation
        }
        subtract_from_temp_variable= {
            temp_COX_landreform_local_support_change = COX_landreform_local_infrastructure
        }
        multiply_temp_variable = {
            temp_COX_landreform_local_support_change = COX_landreform_local_support_increase_factor
        }
        add_to_variable = {
            COX_landreform_local_support = temp_COX_landreform_local_support_change
        }
    }
    else_if = {
        limit = {
            check_variable = { COX_landreform_local_infrastructure < COX_landreform_local_infrastructure_expectation }
        }
        set_temp_variable = {
            temp_COX_landreform_local_support_change = COX_landreform_local_infrastructure
        }
        subtract_from_temp_variable= {
            temp_COX_landreform_local_support_change = COX_landreform_local_infrastructure_expectation
        }
        multiply_temp_variable = {
            temp_COX_landreform_local_support_change = COX_landreform_local_support_decrease_factor
        }
        add_to_variable = {
            COX_landreform_local_support = temp_COX_landreform_local_support_change
        }
    }
}

###GUI相关###
COX_landreform_GUI_get_decisions = {
    clear_array = COX_landreform_GUI_dec_array #GUI用
    if = {
        limit = {
            check_variable = {
                COX_landreform_selected_region = 617
            }
        }
        set_temp_variable = {
            temp_loop_low = 101
        }

        set_temp_variable = {
            temp_loop_up = 118
        }
    }
    else_if = {
        limit = {
            check_variable = {
                COX_landreform_selected_region = 957
            }
        }
        set_temp_variable = {
            temp_loop_low = 201
        }

        set_temp_variable = {
            temp_loop_up = 228
        }
    }
    else_if = {
        limit = {
            check_variable = {
                COX_landreform_selected_region = 618
            }
        }
        set_temp_variable = {
            temp_loop_low = 301
        }

        set_temp_variable = {
            temp_loop_up = 322
        }
    }
    else_if = {
        limit = {
            check_variable = {
                COX_landreform_selected_region = 619
            }
        }
        set_temp_variable = {
            temp_loop_low = 401
        }

        set_temp_variable = {
            temp_loop_up = 427
        }
    }
    for_loop_effect = {
        start = temp_loop_low
        end = temp_loop_up
        value = i

        meta_effect = {
            text = {
                if = {
                    limit = { COX_landreform_dec_[TOKEN]_visible_trigger = yes }
                    add_to_array = {COX_landreform_GUI_dec_array = i}
                }
            }
            TOKEN = "[?i]"
        }
    }
}

COX_landreform_dec_click_effect = {

    }
}
