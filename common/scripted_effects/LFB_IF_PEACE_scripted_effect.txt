PEACE_at_CPC_change = {
	add_to_variable = { PEACE_at_CPC = PEACE_at_change }
	clamp_variable = {
		var = PEACE_at_CPC
		min = 0
		max = 5
	}
	custom_effect_tooltip = "AT_CHANGE_CPC"
}

PEACE_at_KMT_change = {
	add_to_variable = { PEACE_at_KMT = PEACE_at_change }
	clamp_variable = {
		var = PEACE_at_KMT
		min = 0
		max = 5
	}
	custom_effect_tooltip = "AT_CHANGE_KMT"
}

COX_NN_END = {
    if = {
        limit = {
            OR = {
                AND = {
                     check_variable = { PEACE_at_CPC > 4}
                     check_variable = { PEACE_at_KMT > 4}
                      }
                AND = {
                    check_variable = { PEACE_at_CPC = 4}
                    check_variable = { PEACE_at_KMT = 4}
                }
            }
        }
        country_event = {
            id = LFBUG.10
            days = 3
        }
    }
    if = {
        limit = {
            OR = {
                check_variable = { PEACE_at_CPC < 4}
                check_variable = { PEACE_at_KMT < 4}
            }
        }
        set_country_flag = COX_Nanking_Negotiations_Failed_flag
    }
    custom_effect_tooltip = "COX_NN_END_string"
}
