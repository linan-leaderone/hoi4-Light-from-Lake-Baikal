                        }
                    }
                    
                }
            }
            AND = {
                is_in_array = {

                    value = 201
                }
                NOT = {
                    OR = {
                        is_in_array = {
                            array = COX_landreform_dec_done_array
                            value = 214
                        }
                        is_in_array = {
                            array = COX_landreform_dec_done_array
                            value = 225
                        }
                        is_in_array = {
                            array = COX_landreform_dec_done_array
                            value = 227
                        }
                    }
                    
                }
            }
            AND = {
                is_in_array = {
                    array = COX_landreform_dec_done_array
                    value = 301
                }
                NOT = {
                    is_in_array = {
                        array = COX_landreform_dec_done_array
                        value = 318
                    }
                }
            }
            AND = {
                is_in_array = {
                    array = COX_landreform_dec_done_array
                    value = 401
                }
                NOT = {
                    OR = {
