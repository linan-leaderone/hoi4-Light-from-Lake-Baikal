country_event = {#当地暴乱-中央
	id = COX_Landreform.321
	title = COX_Landreform.321.t
	desc = COX_Landreform.321.d
	is_triggered_only = yes

	option = {#放弃
		name = COX_Landreform.321.a
		custom_effect_tooltip = COX_landreform_unrest_event_local_failure_tt
		hidden_effect = {
			617 = {
				add_to_variable = {
					COX_landreform_progress = -5
				}
				COX_landreform_local_unrest_decrease_ultra_ultra = yes
			}
			COX_landreform_refresh = yes
		}

		
	}
	option = {#尝试镇压
		name = COX_Landreform.321.b
		trigger = {
			custom_trigger_tooltip = {
				check_variable = {
					var = COX_landreform_militia
					value = 617.COX_landreform_local_militia_demand
					compare = greater_than_or_equals
				}
				tooltip = COX_landreform_unrest_event_militia_ckeck_tt
			}
			
		}
		custom_effect_tooltip = COX_landreform_unrest_event_militia_use_617_tt
		hidden_effect = {
			if = {
				limit = {
					check_variable = {
						var = COX_landreform_militia
						value = 617.COX_landreform_local_militia_demand
						compare = greater_than_or_equals
					}
				}
				subtract_from_variable = {
					COX_landreform_militia = 617.COX_landreform_local_militia_demand
				}
				617 = {
					COX_landreform_local_unrest_decrease_ultra_ultra = yes
				}
			}
			else = {
				617 = {
					add_to_variable = {
						COX_landreform_progress = -5
					}
					COX_landreform_local_unrest_decrease_ultra_ultra = yes
				}
			}
			COX_landreform_refresh = yes
		}
	}
}
